## TypeScript

Zero `any`. Strict mode on.

```ts
// CORRECT
interface Experiment {
  id: string
  name: string
  status: ExperimentStatus
  trafficPercent: number
  createdAt: string
}

type ExperimentStatus = 'draft' | 'running' | 'paused' | 'completed'

// WRONG
const exp: any = fetchExperiment()
const status = exp.status as string
```

- `interface` for object shapes; `type` for unions, intersections, aliases
- `as const` for static lookup tables, never for type coercion
- No `!` non-null assertions — handle the null/undefined case explicitly
- All function parameters and return types explicit; infer only local variables
- API types come from `types/api.d.ts` (generated — see below), never written by hand

### API Types from OpenAPI spec

Generate types automatically from the backend spec instead of writing them by hand.
This prevents type drift between backend and frontend.

```bash
# package.json
"generate:types": "openapi-typescript ../backend/api/_build/openapi.yml -o types/api.d.ts"
```

Run `pnpm generate:types` after every backend spec change. The output file is committed to git.
`types/index.ts` re-exports and aliases the generated types for convenience:

```ts
// types/index.ts
export type { Experiment, Flag, Variant } from './api.d.ts'
export type ExperimentStatus = components['schemas']['ExperimentStatus']
```

## Providers and Root Layout

`QueryClientProvider` and `NuqsAdapter` are Client Components and cannot live in a Server Component layout directly.
Wrap them in `app/providers.tsx`:

```tsx
// app/providers.tsx
'use client'

import { QueryClient, QueryClientProvider } from '@tanstack/react-query'
import { NuqsAdapter } from 'nuqs/adapters/next/app'
import { useState } from 'react'

export function Providers({ children }: { children: React.ReactNode }) {
  // useState ensures each request gets its own QueryClient (SSR safety)
  const [queryClient] = useState(() => new QueryClient({
    defaultOptions: {
      queries: { staleTime: 30_000, retry: 1 },
    },
  }))

  return (
    <QueryClientProvider client={queryClient}>
      <NuqsAdapter>{children}</NuqsAdapter>
    </QueryClientProvider>
  )
}
```

```tsx
// app/layout.tsx — Server Component
import { Inter } from 'next/font/google'
import { Toaster } from '@/components/ui/sonner'
import { Providers } from './providers'

const inter = Inter({ subsets: ['latin'], variable: '--font-sans' })

export default function RootLayout({ children }: { children: React.ReactNode }) {
  return (
    <html lang="en" className={inter.variable}>
      <body>
        <Providers>{children}</Providers>
        <Toaster />
      </body>
    </html>
  )
}
```

## Auth Guard — middleware.ts

Firebase tokens are verified at the edge in `middleware.ts`. This runs before any page renders.

```ts
// middleware.ts
import { NextRequest, NextResponse } from 'next/server'

const PUBLIC_PATHS = ['/login']

export function middleware(req: NextRequest) {
  const { pathname } = req.nextUrl
  const token = req.cookies.get('firebase-token')?.value

  const isPublic = PUBLIC_PATHS.some((p) => pathname.startsWith(p))

  if (!isPublic && !token) {
    return NextResponse.redirect(new URL('/login', req.url))
  }

  if (isPublic && token) {
    return NextResponse.redirect(new URL('/projects', req.url))
  }

  return NextResponse.next()
}

export const config = {
  matcher: ['/((?!_next|favicon.ico|.*\\..*).*)'],
}
```

Token verification (full signature check) happens in the API client — middleware only checks presence.
The Firebase client SDK writes the token to a cookie on login:

```ts
// lib/auth.ts
import { onIdTokenChanged } from 'firebase/auth'

onIdTokenChanged(auth, async (user) => {
  if (user) {
    const token = await user.getIdToken()
    document.cookie = `firebase-token=${token}; path=/; SameSite=Strict`
  } else {
    document.cookie = 'firebase-token=; path=/; max-age=0'
  }
})
```

## Server vs Client Components

Default to **Server Components**. Add `'use client'` only when needed:
- `useState` / `useReducer`
- `useEffect`
- browser APIs
- event handlers
- React Query hooks

Push `'use client'` as deep as possible — wrap only the interactive leaf, not the whole page.

```tsx
// app/(dashboard)/experiments/[experimentId]/page.tsx — Server Component
export default async function ExperimentPage({ params }: { params: { experimentId: string } }) {
  return (
    <div>
      <ExperimentHeader experimentId={params.experimentId} />
      <ExperimentDetailClient experimentId={params.experimentId} />
    </div>
  )
}
```

```tsx
// components/experiments/experiment-detail-client.tsx
'use client'

export function ExperimentDetailClient({ experimentId }: { experimentId: string }) {
  const { data, isLoading } = useExperiment(experimentId)
  ...
}
```

### Size limit

If a component exceeds ~150 lines or has more than one concern, split it.
A component either holds state OR renders — not both.

```tsx
// WRONG: fetches and renders in one component
export function ExperimentsList({ projectId }: { projectId: string }) {
  const { data } = useExperiments(projectId)
  return <ul>{data?.map(...)}</ul>
}

// CORRECT: split concerns
export function ExperimentsSection({ projectId }: { projectId: string }) {
  const { data, isLoading, isError } = useExperiments(projectId)
  if (isLoading) return <ExperimentsListSkeleton />
  if (isError) return <ErrorState />
  if (!data?.length) return <EmptyState resource="experiments" />
  return <ExperimentsList experiments={data} />
}

function ExperimentsList({ experiments }: { experiments: Experiment[] }) {
  return <ul>{experiments.map((e) => <ExperimentCard key={e.id} experiment={e} />)}</ul>
}
```

## Styling

`cn()` from `lib/utils.ts` for conditional classes. No inline `style={{}}` unless for dynamic values Tailwind cannot express.

```tsx
import { cn } from '@/lib/utils'

// CORRECT
<div className={cn('rounded-lg border p-4', isActive && 'border-primary bg-primary/5')} />

// WRONG — string concat breaks Tailwind IntelliSense purging
<div className={`rounded-lg ${isActive ? 'border-primary' : ''}`} />

// WRONG — inline style for something Tailwind can do
<div style={{ borderRadius: '8px' }} />
```

- Use design tokens (`text-muted-foreground`, `bg-card`, `border`) — never hardcode colours
- Mobile-first responsive: `sm:`, `md:`, `lg:` prefixes
- Never `@apply` in CSS — compose in JSX
- Never edit `components/ui/` directly — customise via `className` prop

## Data Fetching — TanStack Query v5

### Query key factory

All keys in `lib/query-keys.ts`. Never inline key arrays in components.

```ts
// lib/query-keys.ts
export const queryKeys = {
  projects: {
    all: () => ['projects'] as const,
    list: () => [...queryKeys.projects.all(), 'list'] as const,
    detail: (id: string) => [...queryKeys.projects.all(), id] as const,
  },
  experiments: {
    all: () => ['experiments'] as const,
    byProject: (projectId: string) => [...queryKeys.experiments.all(), projectId] as const,
    detail: (id: string) => [...queryKeys.experiments.all(), 'detail', id] as const,
    analytics: (id: string) => [...queryKeys.experiments.detail(id), 'analytics'] as const,
  },
  flags: {
    all: () => ['flags'] as const,
    byProject: (projectId: string) => [...queryKeys.flags.all(), projectId] as const,
  },
} as const
```

### Custom hooks — one file per resource

Components never call `useQuery`/`useMutation` directly — always through a custom hook.

```ts
// hooks/use-experiments.ts
import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query'
import { createExperiment, getExperiment, listExperiments, startExperiment } from '@/lib/api/experiments'
import { queryKeys } from '@/lib/query-keys'

export function useExperiments(projectId: string) {
  return useQuery({
    queryKey: queryKeys.experiments.byProject(projectId),
    queryFn: () => listExperiments(projectId),
  })
}

export function useExperiment(id: string) {
  return useQuery({
    queryKey: queryKeys.experiments.detail(id),
    queryFn: () => getExperiment(id),
    enabled: Boolean(id),
  })
}

export function useStartExperiment() {
  const queryClient = useQueryClient()
  return useMutation({
    mutationFn: startExperiment,
    onSuccess: (updated) => {
      queryClient.setQueryData(queryKeys.experiments.detail(updated.id), updated)
      queryClient.invalidateQueries({ queryKey: queryKeys.experiments.all() })
    },
  })
}
```

### All three states are explicit

```tsx
const { data, isLoading, isError, error } = useExperiments(projectId)

if (isLoading) return <ExperimentsListSkeleton />
if (isError) return <ErrorState message={error.message} />
if (!data?.length) return <EmptyState resource="experiments" />

return <ExperimentsList experiments={data} />
```

## URL State — nuqs

Use `nuqs` for filter/pagination state that belongs in the URL (shareable, back-button friendly).
Never put filters in `useState` — they disappear on refresh.

```ts
// app/(dashboard)/experiments/page.tsx
import { parseAsStringEnum, useQueryState } from 'nuqs'

const statusParser = parseAsStringEnum<ExperimentStatus>(['draft', 'running', 'paused', 'completed'])

export default function ExperimentsPage() {
  const [status, setStatus] = useQueryState('status', statusParser)

  // ?status=running is preserved in URL, back button works correctly
  return <ExperimentFilters status={status} onStatusChange={setStatus} />
}
```

Rule of thumb: if a user would expect the filter to persist on refresh or be shareable via URL → `nuqs`. If it's transient UI state (modal open, hover) → `useState`.

## API Client

```ts
// lib/api/client.ts
import { auth } from '@/lib/auth'

export class ApiError extends Error {
  constructor(
    public readonly status: number,
    message: string,
    public readonly code?: string,
  ) {
    super(message)
    this.name = 'ApiError'
  }
}

async function request<T>(path: string, init?: RequestInit): Promise<T> {
  const token = await auth.currentUser?.getIdToken()

  const res = await fetch(`${process.env.NEXT_PUBLIC_API_URL}${path}`, {
    ...init,
    headers: {
      'Content-Type': 'application/json',
      ...(token && { Authorization: `Bearer ${token}` }),
      ...init?.headers,
    },
  })

  if (!res.ok) {
    const body = await res.json().catch(() => ({}))
    throw new ApiError(res.status, body.message ?? res.statusText, body.code)
  }

  if (res.status === 204) return undefined as T
  return res.json()
}

export const apiClient = {
  get: <T>(path: string) => request<T>(path),
  post: <T>(path: string, body?: unknown) => request<T>(path, { method: 'POST', body: JSON.stringify(body) }),
  patch: <T>(path: string, body: unknown) => request<T>(path, { method: 'PATCH', body: JSON.stringify(body) }),
  delete: <T>(path: string) => request<T>(path, { method: 'DELETE' }),
}
```

## State — Zustand v5

Zustand for **global client state only** — not server data (that's React Query).
Global client state: authenticated user, UI preferences (sidebar collapsed).

```ts
// store/auth-store.ts
import { create } from 'zustand'
import { useShallow } from 'zustand/shallow'   // v5: useShallow replaces shallow equality fn
import type { User } from 'firebase/auth'

interface AuthState {
  user: User | null
  isLoading: boolean
  setUser: (user: User | null) => void
  setLoading: (v: boolean) => void
}

export const useAuthStore = create<AuthState>((set) => ({
  user: null,
  isLoading: true,
  setUser: (user) => set({ user }),
  setLoading: (isLoading) => set({ isLoading }),
}))

// Selectors — prevent unnecessary re-renders
export const useUser = () => useAuthStore((s) => s.user)
export const useIsAuthenticated = () => useAuthStore((s) => s.user !== null)

// For selecting multiple fields — use useShallow (v5 requirement)
export const useAuthStatus = () =>
  useAuthStore(useShallow((s) => ({ user: s.user, isLoading: s.isLoading })))
```

## Forms

`react-hook-form` + `zod` for all forms. Validation schema is the single source of truth.

```tsx
import { useForm } from 'react-hook-form'
import { zodResolver } from '@hookform/resolvers/zod'
import { z } from 'zod'
import { Form, FormControl, FormField, FormItem, FormLabel, FormMessage } from '@/components/ui/form'

const schema = z.object({
  name: z.string().min(1, 'Name is required').max(100),
  trafficPercent: z.number().int().min(1).max(100),
  hypothesis: z.string().optional(),
})

type FormValues = z.infer<typeof schema>

interface CreateExperimentFormProps {
  onSubmit: (values: FormValues) => void
  isPending: boolean
}

export function CreateExperimentForm({ onSubmit, isPending }: CreateExperimentFormProps) {
  const form = useForm<FormValues>({ resolver: zodResolver(schema) })

  return (
    <Form {...form}>
      <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-4">
        <FormField
          control={form.control}
          name="name"
          render={({ field }) => (
            <FormItem>
              <FormLabel>Name</FormLabel>
              <FormControl>
                <Input {...field} />
              </FormControl>
              <FormMessage />
            </FormItem>
          )}
        />
        <Button type="submit" disabled={isPending}>
          {isPending ? 'Creating…' : 'Create experiment'}
        </Button>
      </form>
    </Form>
  )
}
```

## Error Handling

Mutation errors via `sonner` toast — never `alert()` or `window.confirm()`.

```tsx
import { toast } from 'sonner'
import { ApiError } from '@/lib/api/client'

const { mutate: startExp, isPending } = useStartExperiment()

function handleStart() {
  startExp(experimentId, {
    onSuccess: () => toast.success('Experiment started'),
    onError: (err) =>
      toast.error(err instanceof ApiError ? err.message : 'Something went wrong'),
  })
}
```

For async flows (create + navigate), use the `toast.promise` helper:

```ts
toast.promise(createExperimentAsync(values), {
  loading: 'Creating experiment…',
  success: 'Experiment created',
  error: (err) => (err instanceof ApiError ? err.message : 'Failed to create'),
})
```

## Miscellaneous

- All imports use `@/` alias — never relative paths (`../../`)
- No `console.log` in committed code
- Extract status/role values to `as const` objects, not string literals in JSX conditions
